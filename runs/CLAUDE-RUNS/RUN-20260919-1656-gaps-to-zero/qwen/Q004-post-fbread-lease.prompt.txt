# TASK Q004: post-fbread-lease
## Brief
POST.COMPOSITE runs after the raster has drained and before the frame is published, inside the
render engine's framebuffer LEASE. It uses the render engine's own memory client, ENGINE0, and
the memory guard admits ENGINE0 only while `fb_writer == 1` (the render holds the lease). Three
requesters share ENGINE0 inside `zhao_post_lease`: the raster's writer, the post source reader
`zhao_post_fbread`, and the post write-back/echo. `zhao_post_fbread` reads the finished back
buffer in raster order.

## Questions
1. Can `zhao_post_lease` grant ENGINE0 to post's reader or writer while the raster still has writes
   in flight (i.e. before the raster has DRAINED), so post reads pixels that are not final?
2. Can a response beat for one requester be delivered to another (a mis-steer when two requests
   overlap, or a request issued while a response is still streaming)?
3. Does `zhao_post_fbread` read EXACTLY the frame: every pixel once, in raster order, correct at
   the row stride and at the last row and column, for both 1-view and 2-view (Duo) geometry? Check
   the address arithmetic for widths and overflow.
4. Backpressure: if post's consumer stalls mid-row, is any beat lost or duplicated?
5. Any counter or guard that cannot fire (both operands behind one enable, or a condition that cannot arise)?

   68| // ---------------------------------------------------------------------------

   70| // 512 bytes/row, view 1 at logical row 192. So the reader walks ONE tall pass
   71| // (views * h rows from the slot base, no view offset, no multiplier), and the
   72| // compositor and the echo run one pass per view: the compositor stops taking
   73| // source pixels at its view's last pixel, so the reader's next pixel simply
   74| // waits for the next pass. Post's write-back labels view 1's rows `y + h`.
   75| //
   76| // COST (estimated, UNMEASURED): the share (~90 ALM), the sequencer and the
   77| // retire FIFO (~60 ALM), plus the reader and the echo (their own headers).
   78| `default_nettype none

   80| module zhao_post_lease
   81|   import zhao_pkg::*;
   82| #(
   83|     parameter int unsigned XW = 9,
   84|     parameter int unsigned YW = 8

   99|     input  var logic [YW-1:0] frame_h_i,

  109|     output var logic          out_ready_o,

  121|     output var logic [15:0]        fbw_px_rgb_o,
  122|     output var logic signed [11:0] fbw_px_x_o,
  123|     output var logic signed [11:0] fbw_px_y_o,
  124|     output var logic               fbw_px_last_o,

  126|     // ---- requester 0: RASTER.FBWRITE's guard master and write channel -------
  127|     input  var zhao_guard_req_t fbw_req_i,

  129|     input  var logic [63:0]     fbw_wdata_i,
  130|     input  var logic            fbw_wvalid_i,
  131|     output var logic            fbw_wready_o,
  132|     input  var logic            fbw_wlast_i,
  133|     output var logic [7:0]      fbw_retire_o,
  134| 
  135|     // ---- ENGINE0 towards MEM.GUARD, the write queue and the credit stream ---
  136|     output var zhao_guard_req_t e0_req_o,
  137|     input  var zhao_guard_rsp_t e0_rsp_i,
  138|     output var logic [63:0]     e0_wdata_o,
  139|     output var logic            e0_wvalid_o,
  140|     input  var logic            e0_wready_i,
  141|     output var logic            e0_wlast_o,
  142|     input  var logic            e0_beat_valid_i,  // ENGINE0 read beats, routed
  143|     input  var logic [63:0]     e0_beat_data_i,
  144|     input  var logic            e0_beat_last_i,
  145|     input  var logic [7:0]      e0_credits_i,     // client_rsp[ENGINE0].credits
  146| 
  147|     // ---- evidence -------------------------------------------------------------
  148|     output var logic          busy_o,            // armed or running: do not publish
  149|     output var logic [31:0]   passes_o,          // compositor passes completed
  150|     output var logic [31:0]   frames_o,          // frames fully post-processed
  151|     output var logic          fault_o,           // sticky: a refused source read
  152|     output var logic [31:0]   src_reads_o,
  153|     output var logic [31:0]   src_pixels_o,
  154|     output var logic [31:0]   retire_unowned_o,  // tripwire, must read 0
  155|     output var logic [31:0]   share_contention_o,
  156|     output var logic [31:0]   echo_passes_complete_o,
  157|     output var logic [31:0]   echo_passes_torn_o,
  158|     output var logic [31:0]   echo_pixels_written_o,
  159|     output var logic [31:0]   echo_pixels_dropped_o,
  160|     output var logic          echo_fault_o
  161| );
  162| 
  163|   localparam int unsigned RYW = YW + 1;          // the reader's tall Duo pass
  164| 
  165|   // ==========================================================================
  166|   // THE SEQUENCER
  167|   // ==========================================================================
  168|   typedef enum logic [2:0] {
  169|     S_IDLE  = 3'd0,   // no render frame to post-process
  170|     S_ARMED = 3'd1,   // frame_end seen; waiting for the raster to drain
  171|     S_PASS  = 3'd2,   // a compositor pass is running
  172|     S_SETTLE= 3'd3,   // the pass's last pixel is out; wait for retirement
  173|     S_DONE  = 3'd4    // every view post-processed; hold until the next frame
  174|   } sstate_e;
  175| 
  176|   sstate_e       st_q;
  177|   logic          view_q;
  178|   logic          start_c;
  179|   logic          last_out_seen_q;
  180| 
  181|   // The view the NEXT pass opens: the first pass of a frame is view 0, the
  182|   // second (Duo only) is view 1.
  183|   logic          next_view_c;
  184|   assign next_view_c = (st_q == S_SETTLE);

  186|   // The share's view of each requester, declared here because the sequencer,
  187|   // the write-order mask and the retire ledger all read it.
  188|   zhao_guard_req_t [2:0] sh_req;
  189|   zhao_guard_rsp_t [2:0] sh_rsp;
  190|   logic [2:0][6:0]       sh_req_len_q;     // length when the share took it
  191| 
  192|   // The raster is finished with the slot: every tile resolved, nothing on the
  193|   // pixel port, every raster word retired.
  194|   logic raster_done_c;
  195|   assign raster_done_c = raster_quiet_i && !raster_px_i && fbw_drained_i;
  196| 
  197|   logic echo_busy, rd_busy, rd_done_unused, rd_fault;
  198| 
  199|   assign start_c = ((st_q == S_ARMED) && raster_done_c && lease_live_i)
  200|                 || ((st_q == S_SETTLE) && fbw_drained_i && !echo_busy
  201|                     && duo_i && !view_q);
  202| 
  203|   assign pass_start_o = start_c;
  204|   assign view_o       = view_q;
  205|   assign phase_post_o = (st_q == S_PASS) || (st_q == S_SETTLE) || (st_q == S_DONE);
  206|   assign busy_o       = (st_q == S_ARMED) || (st_q == S_PASS) || (st_q == S_SETTLE);
  207| 
  208|   // ==========================================================================
  209|   // THE SOURCE -- one tall read for every view of the frame
  210|   // ==========================================================================
  211|   zhao_guard_req_t rd_req;
  212|   zhao_guard_rsp_t rd_rsp;
  213|   logic            rd_beat_v;
  214|   logic [63:0]     rd_beat_d;
  215|   logic [RYW-1:0]  rd_h_c;
  216|   assign rd_h_c = duo_i ? (RYW'(frame_h_i) << 1) : RYW'(frame_h_i);
  217| 
  218|   logic [31:0] rd_overflow_unused;
  219|   zhao_post_fbread #(.XW(XW), .YW(RYW), .FIFO_BEATS(16)) u_source (
  220|     .clk(clk), .rst_n(rst_n),
  221|     // ONE read per FRAME: only the first view's start opens it.
  222|     .start_i (start_c && (st_q == S_ARMED)),
  223|     .origin_i(fb_base_i), .stride_i(fb_stride_i),
  224|     .w_i(frame_w_i), .h_i(rd_h_c),
  225|     .guard_req_o(rd_req), .guard_rsp_i(rd_rsp),
  226|     .beat_valid_i(rd_beat_v), .beat_data_i(rd_beat_d),
  227|     .px_valid_o(src_valid_o), .px_ready_i(src_ready_i), .px_rgb_o(src_rgb_o),
  228|     .busy_o(rd_busy), .done_o(rd_done_unused), .fault_o(rd_fault),
  229|     .reads_o(src_reads_o), .pixels_o(src_pixels_o), .overflow_o(rd_overflow_unused)
  230|   );
  231|   assign fault_o = rd_fault;
  232| 
  233|   // ==========================================================================
  234|   // THE WRITE-BACK -- the compositor's output onto RASTER.FBWRITE's port
  235|   // ==========================================================================
  236|   // Canvas row = view-local row, plus h for view 1 (the stored Duo surface).
  237|   logic [RYW-1:0] out_row_c;
  238|   assign out_row_c = view_q ? (RYW'(out_y_i) + RYW'(frame_h_i)) : RYW'(out_y_i);
  239| 
  240|   assign fbw_px_valid_o = out_valid_i && (st_q == S_PASS);
  241|   assign out_ready_o    = fbw_px_ready_i && (st_q == S_PASS);
  242|   assign fbw_px_rgb_o   = out_rgb_i;
  243|   assign fbw_px_x_o     = 12'($signed({1'b0, out_x_i}));
  244|   assign fbw_px_y_o     = 12'($signed({1'b0, out_row_c}));
  245|   assign fbw_px_last_o  = out_last_i;
  246| 
  247|   logic out_fire_c;
  248|   assign out_fire_c = out_valid_i && out_ready_o;
  249| 
  250|   // ==========================================================================
  251|   // POST.ECHO -- on the compositor's tap, qualified by the output handshake
  252|   // ==========================================================================
  253|   zhao_guard_req_t ec_req;
  254|   zhao_guard_rsp_t ec_rsp;
  255|   logic [63:0]     ec_wdata;
  256|   logic            ec_wvalid, ec_wready, ec_wlast;
  257|   logic [7:0]      ec_retire;
  258|   logic            ec_complete_unused;
  259| 
  260|   zhao_post_echo #(.XW(XW), .YW(YW), .SKID(256)) u_echo (
  261|     .clk(clk), .rst_n(rst_n),
  262|     .pass_start_i(start_c), .view_i(start_c ? next_view_c : view_q),
  263|     .w_i(frame_w_i), .h_i(frame_h_i),
  264|     // The raw tap repeats a stalled beat; the ACCEPTED beat happens once.
  265|     .tap_valid_i(echo_valid_i && out_fire_c), .tap_rgb_i(echo_rgb_i),
  266|     .tap_x_i(out_x_i), .tap_y_i(out_y_i),
  267|     .guard_req_o(ec_req), .guard_rsp_i(ec_rsp),
  268|     .guard_wdata_o(ec_wdata), .guard_wvalid_o(ec_wvalid),
  269|     .guard_wready_i(ec_wready), .guard_wlast_o(ec_wlast),
  270|     .retire_words_i(ec_retire),
  271|     .busy_o(echo_busy), .pass_complete_o(ec_complete_unused),
  272|     .passes_complete_o(echo_passes_complete_o), .passes_torn_o(echo_passes_torn_o),
  273|     .pixels_written_o(echo_pixels_written_o), .pixels_dropped_o(echo_pixels_dropped_o),
  274|     .fault_o(echo_fault_o)
  275|   );
  276| 
  277| 
  278|   // ==========================================================================
  279|   // ENGINE0 -- three logical requesters, one client (zhao_mem_share_n)
  280|   // ==========================================================================
  281|   // WRITE-DATA ORDER: from the moment the share TAKES a write until that
  282|   // write's last data beat (or its refusal), the other writer's request is
  283|   // withheld. Registered at the take, which is early enough: the share takes
  284|   // one requester per idle cycle and is busy until the verdict, so no second
  285|   // write can be taken before this mask is up -- and registered, so it forms
  286|   // no combinational path back through the share's ready.
  287|   logic            wpend_q;
  288|   logic            wown_q;                 // 0 = FBWRITE, 1 = ECHO
  289|   logic            wpend_c;
  290|   assign wpend_c = wpend_q;
  291| 
  292|   always_comb begin
  293|     sh_req[0] = fbw_req_i;
  294|     sh_req[1] = rd_req;
  295|     sh_req[2] = ec_req;
  296|     if (wpend_c) begin
  297|       sh_req[0].valid = 1'b0;
  298|       sh_req[2].valid = 1'b0;
  299|     end
  300|   end
  301| 
  302|   logic [2:0]       sh_bv, sh_bl_unused;
  303|   logic [63:0]      sh_bd;
  304|   logic [2:0][31:0] sh_jobs_unused;
  305|   logic [31:0]      sh_denied_unused, sh_short_unused, sh_long_unused, sh_unowned_unused;
  306| 
  307|   zhao_mem_share_n #(.N(3), .CLIENT_ID(2), .FORCE_READ(1'b0)) u_engine0_share (
  308|     .clk(clk), .rst_n(rst_n),
  309|     .req_i(sh_req), .rsp_o(sh_rsp),
  310|     .beat_valid_o(sh_bv), .beat_data_o(sh_bd), .beat_last_o(sh_bl_unused),
  311|     .m_req_o(e0_req_o), .m_rsp_i(e0_rsp_i),
  312|     .m_beat_valid_i(e0_beat_valid_i), .m_beat_data_i(e0_beat_data_i),
  313|     .m_beat_last_i(e0_beat_last_i),
  314|     .jobs_o(sh_jobs_unused), .denied_o(sh_denied_unused),
  315|     .contention_o(share_contention_o),
  316|     .err_short_o(sh_short_unused), .err_long_o(sh_long_unused),
  317|     .err_unowned_o(sh_unowned_unused)
  318|   );

  320|   // The share is the guard as each requester sees it. A requester whose
  321|   // request is masked sees no ready, which is simply the guard being busy.
  322|   assign fbw_rsp_o = sh_rsp[0];
  323|   assign rd_rsp    = sh_rsp[1];
  324|   assign ec_rsp    = sh_rsp[2];
  325|   assign rd_beat_v = sh_bv[1];
  326|   assign rd_beat_d = sh_bd;
  327| 
  328|   // The one outstanding writer owns the data channel.
  329|   assign e0_wvalid_o  = wown_q ? ec_wvalid : fbw_wvalid_i;
  330|   assign e0_wdata_o   = wown_q ? ec_wdata  : fbw_wdata_i;
  331|   assign e0_wlast_o   = wown_q ? ec_wlast  : fbw_wlast_i;
  332|   assign fbw_wready_o = !wown_q && e0_wready_i;
  333|   assign ec_wready    =  wown_q && e0_wready_i;
  334| 
  335|   // ==========================================================================
  336|   // RETIREMENT ATTRIBUTION -- in order, as the controller retires
  337|   // ==========================================================================
  338|   // {requester, words} pushed at each passed verdict; the credit stream drains
  339|   // the head. Depth 4 covers a request in the guard, one at the arbiter and one
  340|   // retiring, with one to spare.
  341|   localparam int unsigned RQ = 4;
  342|   logic [1:0] rq_own [0:RQ-1];
  343|   logic [5:0] rq_left[0:RQ-1];
  344|   logic [1:0] rq_wp_q, rq_rp_q;
  345|   logic [2:0] rq_n_q;
  346| 
  347|   logic       push_c;
  348|   logic [1:0] push_own_c;
  349|   logic [5:0] push_words_c;
  350|   always_comb begin
  351|     push_c       = 1'b0;
  352|     push_own_c   = 2'd0;
  353|     push_words_c = 6'd0;
  354|     for (int k = 0; k < 3; k++) begin
  355|       if (sh_rsp[k].ok) begin
  356|         push_c       = 1'b1;
  357|         push_own_c   = 2'(k);
  358|         push_words_c = 6'(sh_req_len_q[k] >> 1);   // bytes -> 16-bit words
  359|       end
  360|     end
  361|   end
  362| 
  363|   // The length each requester last offered, captured when the share took it
  364|   // (its ready), because the requester drops the request before the verdict.
  365|   always_ff @(posedge clk or negedge rst_n) begin
  366|     if (!rst_n) sh_req_len_q <= '0;
  367|     else begin
  368|       for (int k = 0; k < 3; k++)
  369|         if (sh_req[k].valid && sh_rsp[k].ready) sh_req_len_q[k] <= sh_req[k].len;
  370|     end
  371|   end
  372| 
  373|   logic       credit_c;
  374|   logic [1:0] head_own_c;
  375|   assign credit_c   = (e0_credits_i != 8'd0);
  376|   assign head_own_c = rq_own[rq_rp_q];
  377| 
  378|   assign fbw_retire_o = (credit_c && (rq_n_q != 3'd0) && (head_own_c == 2'd0)) ? e0_credits_i : 8'd0;
  379|   assign ec_retire    = (credit_c && (rq_n_q != 3'd0) && (head_own_c == 2'd2)) ? e0_credits_i : 8'd0;
  380| 
  381|   always_ff @(posedge clk or negedge rst_n) begin
  382|     if (!rst_n) begin
  383|       rq_wp_q <= '0; rq_rp_q <= '0; rq_n_q <= '0;
  384|       retire_unowned_o <= '0;
  385|       for (int i = 0; i < RQ; i++) begin rq_own[i] <= '0; rq_left[i] <= '0; end
  386|     end else begin
  387|       automatic logic pop = 1'b0;
  388|       if (credit_c) begin
  389|         if (rq_n_q == 3'd0) begin
  390|           retire_unowned_o <= retire_unowned_o + 32'd1;
  391|         end else if (6'(e0_credits_i) >= rq_left[rq_rp_q]) begin
  392|           pop = 1'b1;
  393|           rq_rp_q <= rq_rp_q + 2'd1;
  394|           // A burst's credits belong to ONE request. More than the head is
  395|           // owed means a word was attributed to the wrong requester.
  396|           if (6'(e0_credits_i) != rq_left[rq_rp_q])
  397|             retire_unowned_o <= retire_unowned_o + 32'd1;
  398|         end else begin
  399|           rq_left[rq_rp_q] <= rq_left[rq_rp_q] - 6'(e0_credits_i);
  400|         end
  401|       end
  402|       if (push_c) begin
  403|         rq_own[rq_wp_q]  <= push_own_c;
  404|         rq_left[rq_wp_q] <= push_words_c;
  405|         rq_wp_q          <= rq_wp_q + 2'd1;
  406|       end
  407|       rq_n_q <= rq_n_q + (push_c ? 3'd1 : 3'd0) - (pop ? 3'd1 : 3'd0);
  408|     end
  409|   end
  410| 
  411|   // ==========================================================================
  412|   // THE ONE SEQUENTIAL BLOCK FOR THE PASS
  413|   // ==========================================================================
  414|   always_ff @(posedge clk or negedge rst_n) begin
  415|     if (!rst_n) begin
  416|       st_q            <= S_IDLE;
  417|       view_q          <= 1'b0;
  418|       last_out_seen_q <= 1'b0;
  419|       wpend_q         <= 1'b0;
  420|       wown_q          <= 1'b0;
  421|       passes_o        <= '0;
  422|       frames_o        <= '0;
  423|     end else begin
  424|       // ---- write-data ownership --------------------------------------------
  425|       if (sh_req[0].valid && sh_rsp[0].ready && sh_req[0].write) begin
  426|         wpend_q <= 1'b1; wown_q <= 1'b0;
  427|       end
  428|       if (sh_req[2].valid && sh_rsp[2].ready) begin
  429|         wpend_q <= 1'b1; wown_q <= 1'b1;
  430|       end
  431|       // Released by the owner's last data beat, or by its REFUSAL: a refused
  432|       // write sends no data, and holding the mask would lock out the other
  433|       // writer for good.
  434|       if ((e0_wvalid_o && e0_wready_i && e0_wlast_o)
  435|           || (wown_q ? sh_rsp[2].violation : sh_rsp[0].violation))
  436|         wpend_q <= 1'b0;
  437| 
  438|       // ---- the pass ---------------------------------------------------------
  439|       if (out_fire_c && out_last_i) last_out_seen_q <= 1'b1;
  440| 
  441|       case (st_q)
  442|         S_IDLE: begin
  443|           if (frame_end_i && lease_live_i) st_q <= S_ARMED;
  444|         end
  445|         S_ARMED: begin
  446|           if (start_c) begin
  447|             st_q            <= S_PASS;
  448|             view_q          <= 1'b0;
  449|             last_out_seen_q <= 1'b0;
  450|           end
  451|         end
  452|         S_PASS: begin
  453|           if (last_out_seen_q || (out_fire_c && out_last_i)) st_q <= S_SETTLE;
  454|         end
  455|         S_SETTLE: begin
  456|           if (start_c) begin                       // Duo: view 1
  457|             st_q            <= S_PASS;
  458|             view_q          <= 1'b1;
  459|             last_out_seen_q <= 1'b0;
  460|             passes_o        <= passes_o + 32'd1;
  461|           end else if (fbw_drained_i && !echo_busy) begin
  462|             st_q     <= S_DONE;
  463|             passes_o <= passes_o + 32'd1;
  464|             frames_o <= frames_o + 32'd1;
  465|           end
  466|         end
  467|         S_DONE: ;
  468|         default: st_q <= S_IDLE;
  469|       endcase
  470| 
  471|       // A newly admitted render frame ends the previous post phase: the shell
  472|       // gives RASTER.FBWRITE back to the raster from here.
  473|       if (frame_admit_i) st_q <= S_IDLE;
  474|     end
  475|   end
  476| 
  477| 
  478|   logic unused_c;
  479|   assign unused_c = ^{rd_busy, rd_done_unused, rd_overflow_unused, ec_complete_unused,
  480|                       sh_bv[0], sh_bv[2], sh_bl_unused, sh_jobs_unused, sh_denied_unused,
  481|                       sh_short_unused, sh_long_unused, sh_unowned_unused};

  483| endmodule : zhao_post_lease
